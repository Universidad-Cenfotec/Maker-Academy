## Contenido

### Diferencias con la IdeaBoard

El Uno trabaja a 5 V: el sensor no necesita divisor. Las luces van en pines con PWM (3, 5 y 6).

<img width="380" alt="Arduino Uno" src="https://upload.wikimedia.org/wikipedia/commons/3/38/Arduino_Uno_-_R3.jpg" />

### Conexiones

| Componente | Conexión |
|---|---|
| Grupos de luces 1, 2 y 3 | Pines 3, 5 y 6, cada LED con su resistencia de 220 Ω a GND |
| HC-SR05 | TRIG al 12, ECHO al 11, VCC a 5V, GND a GND |

### Código

```cpp
// Proyecto 7: Volcan Poas vivo (Arduino Uno)
// Arduino Uno: LEDs en pines PWM 3, 5 y 6; el sensor trabaja a 5 V sin divisor
// Envia "ERUPCION" por el cable USB para que la proyeccion cambie.
const int LUCES[3] = {3, 5, 6};      // cada pin: 2 LEDs con su resistencia de 220 Ohm
const int PIN_TRIG = 12;
const int PIN_ECHO = 11;
const int DISTANCIA_CM = 80;
const unsigned long DURACION = 15000;
const unsigned long ESPERA = 20000;
unsigned long ultima = 0;
bool primera = true;

long medirDistanciaCm() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  long t = pulseIn(PIN_ECHO, HIGH, 30000);
  if (t == 0) return 999;
  return t / 58;
}

void calma() {
  float t = millis() / 1000.0;
  for (int i = 0; i < 3; i++) analogWrite(LUCES[i], 12 + 12 * sin(t * 1.2 + i));
}

void erupcion() {
  Serial.println("ERUPCION");
  unsigned long fin = millis() + DURACION;
  while (millis() < fin) {
    for (int i = 0; i < 3; i++) analogWrite(LUCES[i], random(80, 256));
    delay(random(30, 120));
  }
  Serial.println("CALMA");
}

void setup() {
  Serial.begin(115200);
  for (int i = 0; i < 3; i++) pinMode(LUCES[i], OUTPUT);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  Serial.println("Volcan listo");
}

void loop() {
  calma();
  if ((primera || millis() - ultima > ESPERA) && medirDistanciaCm() < DISTANCIA_CM) {
    erupcion();
    ultima = millis();
    primera = false;
  }
  delay(50);
}
```

### Prueba de funcionamiento

En el monitor serie aparecen ERUPCION y CALMA. Ciérrelo antes de conectar la proyección.

### Limitaciones y soluciones

- Cada pin del Uno entrega hasta unos 20 mA: dos LEDs con 220 Ω por pin están dentro del límite.
