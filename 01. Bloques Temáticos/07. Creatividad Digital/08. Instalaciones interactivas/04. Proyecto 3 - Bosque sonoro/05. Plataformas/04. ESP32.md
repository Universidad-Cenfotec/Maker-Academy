## Contenido

### Diferencias con la IdeaBoard

El ESP32 genérico usa los mismos pines y el mismo comportamiento que la IdeaBoard, pero no trae LED RGB ni filas con V+ para cada pin: el sensor y la tira se alimentan desde el pin VIN (5 V) de la placa.

<img width="380" alt="placa ESP32 DevKit" src="https://upload.wikimedia.org/wikipedia/commons/thumb/2/20/ESP32_Espressif_ESP-WROOM-32_Dev_Board.jpg/960px-ESP32_Espressif_ESP-WROOM-32_Dev_Board.jpg" />

### Conexiones

| Componente | Pin del ESP32 |
|---|---|
| Hojas 1 a 4 | 4, 27, 32 y 33 (pines táctiles) |
| Buzzer pasivo | 5 y GND |
| Tira WS2812B | DIN al 18, 5V a VIN, GND a GND |
| HC-SR05 | TRIG al 26; ECHO por el divisor al 35; VCC a VIN |

### Programación

1. En Arduino IDE agregue las placas ESP32 desde el gestor de tarjetas e instale la librería **Adafruit NeoPixel**.
2. Seleccione **ESP32 Dev Module** y el puerto.

### Código

```cpp
// Proyecto 3: Bosque sonoro (ESP32 generico, Arduino IDE)
// Las hojas usan los pines tactiles del ESP32: no necesitan resistencias.
#include <Adafruit_NeoPixel.h>

const int PIN_BUZZER = 5;
const int PIN_TIRA = 18;
const int PIN_TRIG = 26;
const int PIN_ECHO = 35;          // con divisor 1 kOhm / 2 kOhm (el sensor da 5 V)
const int PINES_HOJAS[4] = {4, 27, 32, 33};
const char* NOMBRES[4] = {"yiguirro", "rana", "mono congo", "tucan"};
const int NUM_LEDS = 30;
const float SENSIBILIDAD = 0.15;
const int DIST_DESPIERTO = 120;
const int DIST_TORMENTA = 40;
const unsigned long TIEMPO_CERCA = 3000;
const unsigned long DURACION_TORMENTA = 6000;

Adafruit_NeoPixel tira(NUM_LEDS, PIN_TIRA, NEO_GRB + NEO_KHZ800);
int base[4];
bool antes[4] = {false, false, false, false};

enum Estado { DORMIDO, DESPIERTO, TORMENTA };
Estado estado = DORMIDO;
unsigned long inicioCerca = 0;
unsigned long inicioTormenta = 0;
unsigned long toques[6] = {0, 0, 0, 0, 0, 0};
int posToque = 0;

void nota(int f, int ms) {
  if (f > 0) tone(PIN_BUZZER, f, ms);
  delay(ms + 20);
}

void sonarAnimal(int i) {
  if (i == 0) { nota(1320, 120); nota(1570, 120); nota(1320, 120); nota(1050, 250); }
  if (i == 1) { nota(180, 80); nota(0, 60); nota(180, 80); nota(0, 60); nota(220, 150); }
  if (i == 2) { nota(110, 300); nota(130, 300); nota(150, 400); }
  if (i == 3) { nota(700, 60); nota(500, 60); nota(700, 60); nota(500, 100); }
}

uint32_t colorAnimal(int i) {
  if (i == 0) return tira.Color(255, 180, 0);
  if (i == 1) return tira.Color(0, 255, 40);
  if (i == 2) return tira.Color(255, 60, 0);
  return tira.Color(0, 120, 255);
}

long medirDistanciaCm() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  long t = pulseIn(PIN_ECHO, HIGH, 30000);
  if (t == 0) return 999;
  return t / 58;
}

void pintar(int r, int g, int b) {
  tira.fill(tira.Color(r, g, b));
  tira.show();
}

void cambiar(Estado nuevo) {
  if (nuevo != estado) {
    Serial.print("Estado: ");
    Serial.println(nuevo == DORMIDO ? "DORMIDO" : (nuevo == DESPIERTO ? "DESPIERTO" : "TORMENTA"));
    estado = nuevo;
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  tira.begin();
  tira.setBrightness(100);
  for (int i = 0; i < 4; i++) base[i] = touchRead(PINES_HOJAS[i]);  // no tocar al encender
  Serial.println("Bosque sonoro listo");
}

void loop() {
  unsigned long ahora = millis();
  for (int i = 0; i < 4; i++) {
    int valor = touchRead(PINES_HOJAS[i]);
    bool tocada = abs(valor - base[i]) > base[i] * SENSIBILIDAD;
    if (tocada && !antes[i]) {
      Serial.print("Hoja: ");
      Serial.println(NOMBRES[i]);
      toques[posToque] = ahora;
      posToque = (posToque + 1) % 6;
      if (estado == DESPIERTO) {
        int n = NUM_LEDS / 4;
        for (int k = i * n; k < (i + 1) * n; k++) tira.setPixelColor(k, colorAnimal(i));
        tira.show();
        sonarAnimal(i);
      }
    }
    antes[i] = tocada;
  }

  long d = medirDistanciaCm();
  bool seisToques = toques[posToque] != 0 && ahora - toques[posToque] < 10000;
  if (estado == TORMENTA) {
    if (random(100) < 15) {
      pintar(255, 255, 255);
      delay(50);
      nota(random(60, 120), 250);
    }
    pintar(0, 0, random(10, 40));
    if (ahora - inicioTormenta > DURACION_TORMENTA) {
      for (int k = 0; k < 6; k++) toques[k] = 0;
      cambiar(DESPIERTO);
    }
  } else {
    if (d < DIST_TORMENTA) { if (inicioCerca == 0) inicioCerca = ahora; }
    else inicioCerca = 0;
    bool muyCerca = inicioCerca != 0 && ahora - inicioCerca > TIEMPO_CERCA;
    if (muyCerca || seisToques) {
      inicioTormenta = ahora;
      inicioCerca = 0;
      cambiar(TORMENTA);
    } else if (d < DIST_DESPIERTO) {
      if (estado == DORMIDO) { cambiar(DESPIERTO); nota(523, 100); nota(659, 100); nota(784, 150); }
      pintar(40, 90, 10);
    } else {
      cambiar(DORMIDO);
      int v = 10 + 12 * (1 + sin(ahora / 650.0));
      pintar(0, v, v / 3);
    }
  }
  delay(30);
}
```

### Prueba de funcionamiento

Igual que en la guía de la IdeaBoard: dormido lejos, despierto a menos de 120 cm y tormenta a menos de 40 cm por 3 segundos.

### Limitaciones y soluciones

- No toque las hojas al encender: el programa guarda el valor sin tocar en `setup()`.
- El pin 35 solo es de entrada: sirve para ECHO, no para TRIG.
